```python
from dataclasses import dataclass
from typing import Tuple, Dict

@dataclass
class TokenPositionMeasurement:
    """Measure effects at different token positions separately."""
    harm_perception_change: float  # ∂(harm_classification) / ∂v at t_inst
    refusal_behavior_change: float  # ∂(refusal_behavior) / ∂v at t_post-inst
    gram_at_inst: torch.Tensor    # Gradient at instruction position
    gram_at_post: torch.Tensor    # Gradient at generation position

class StratifiedMeasurer:
    """Measurement function supporting token-position analysis."""

    def __init__(self, model, tokenizer, harm_classifier):
        self.model = model
        self.tokenizer = tokenizer
        self.harm_classifier = harm_classifier  # Trained classifier for harm

    def measure_at_position(self,
                           v: torch.Tensor,
                           prompts: List[str],
                           position: str = "t_inst") -> TokenPositionMeasurement:
        """
        Measure refusal strength at specific token position.

        Args:
            v: Direction vector to test
            prompts: List of harmful prompts
            position: "t_inst" (instruction end) or "t_post-inst" (generation start)

        Returns:
            TokenPositionMeasurement with position-specific effects
        """
        v.requires_grad = True

        # Get hook at specific position
        target_position = {"t_inst": -1, "t_post-inst": 0}[position]

        all_hidden_states = []

        def capture_hook(module, input, output):
            # Capture hidden state at target token position
            all_hidden_states.append(output[0][:, target_position, :].detach())

        # Register hook
        hook = self.model.model.layers[-1].register_forward_hook(capture_hook)

        try:
            # Forward pass
            with torch.no_grad():
                for prompt in prompts:
                    inputs = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
                    self.model(**inputs)

            h_inst = torch.stack(all_hidden_states).mean(dim=0)  # [hidden_dim]

            # Measure 1: How much does this direction change harm perception?
            if position == "t_inst":
                harm_logits = self.harm_classifier(h_inst)
                harm_perception = torch.sigmoid(harm_logits)

                # Steer and remeasure
                h_steered = h_inst + v / v.norm()  # Unit step
                harm_logits_steered = self.harm_classifier(h_steered)
                harm_perception_steered = torch.sigmoid(harm_logits_steered)

                harm_change = (harm_perception_steered - harm_perception).mean()
                harm_change.backward()
                gram_inst = v.grad.clone()

                return TokenPositionMeasurement(
                    harm_perception_change=harm_change.item(),
                    refusal_behavior_change=0.0,
                    gram_at_inst=gram_inst,
                    gram_at_post=None
                )

            # Measure 2: How much does this direction change refusal behavior?
            elif position == "t_post-inst":
                # Generate with and without vector
                responses_original = self.model.generate(
                    self.tokenizer(prompts[0], return_tensors="pt")["input_ids"],
                    max_length=50
                )

                # Steer hidden states during generation
                responses_steered = self.model.generate(
                    self.tokenizer(prompts[0], return_tensors="pt")["input_ids"],
                    max_length=50
                    # (Would need custom forward pass to steer at t_post-inst)
                )

                refusal_rate = self.measure_refusal(responses_original)
                refusal_rate_steered = self.measure_refusal(responses_steered)

                refusal_change = refusal_rate_steered - refusal_rate

                return TokenPositionMeasurement(
                    harm_perception_change=0.0,
                    refusal_behavior_change=refusal_change,
                    gram_at_inst=None,
                    gram_at_post=v.grad.clone() if v.grad else None
                )

        finally:
            hook.remove()

    def measure_orthogonality(self,
                             v_harm: torch.Tensor,
                             v_refusal: torch.Tensor) -> float:
        """
        Measure orthogonality between directions.
        Lower = more orthogonal (independent).
        """
        # Normalize
        v_harm_norm = v_harm / v_harm.norm()
        v_refusal_norm = v_refusal / v_refusal.norm()

        # Dot product
        dot_product = (v_harm_norm * v_refusal_norm).sum()

        return dot_product.item()
```

## 2. Dual-Direction Discovery

### Harmfulness Direction Discovery

```python
def discover_harmfulness_direction(model, tokenizer, harm_classifier,
                                  harmful_prompts, harmless_prompts,
                                  device="cuda"):
    """
    Discover the harmfulness direction via clustering analysis.

    Based on 2507.11878: harmfulness is encoded at t_inst position.
    """

    # Collect activations at instruction position
    harmful_acts = []
    harmless_acts = []

    for prompts, acts_list in [
        (harmful_prompts, harmful_acts),
        (harmless_prompts, harmless_acts)
    ]:
        for prompt in prompts:
            inputs = tokenizer(prompt, return_tensors="pt").to(device)

            # Forward pass
            with torch.no_grad():
                outputs = model(**inputs, output_hidden_states=True)
                # Get activation at last token position (t_inst)
                h_inst = outputs.hidden_states[-1][0, -1, :]
                acts_list.append(h_inst)

    harmful_acts = torch.stack(harmful_acts)      # [n_harmful, hidden_dim]
    harmless_acts = torch.stack(harmless_acts)    # [n_harmless, hidden_dim]

    # Method 1: Mean difference (like Arditi et al.)
    harmfulness_direction = harmful_acts.mean(0) - harmless_acts.mean(0)
    harmfulness_direction = harmfulness_direction / harmfulness_direction.norm()

    # Method 2: PCA-based (more robust)
    all_acts = torch.cat([harmful_acts, harmless_acts])
    labels = torch.cat([
        torch.ones(len(harmful_acts)),
        torch.zeros(len(harmless_acts))
    ])

    # Center
    centered = all_acts - all_acts.mean(0)

    # PCA: find direction that best separates harmful vs harmless
    cov = (centered.T @ centered) / len(centered)
    eigvals, eigvecs = torch.linalg.eigh(cov)

    # Use first principal component (or LDA direction)
    harmfulness_direction = eigvecs[:, -1]  # Largest eigenvector

    # Verify: this direction should predict harmful vs harmless
    scores = (all_acts @ harmfulness_direction)
    harmful_score_mean = scores[:len(harmful_acts)].mean()
    harmless_score_mean = scores[len(harmful_acts):].mean()

    print(f"Harm score: {harmful_score_mean:.3f}")
    print(f"Harmless score: {harmless_score_mean:.3f}")
    print(f"Separation: {abs(harmful_score_mean - harmless_score_mean):.3f}")

    return harmfulness_direction


def discover_refusal_direction(model, tokenizer,
                              refused_prompts, accepted_prompts,
                              device="cuda"):
    """
    Discover refusal direction at t_post-inst (generation start).

    This is similar to your existing gradient discovery,
    but stratified to the specific token position.
    """

    refusal_acts = []
    accepted_acts = []

    for prompts, acts_list, label in [
        (refused_prompts, refusal_acts, 1),
        (accepted_prompts, accepted_acts, 0)
    ]:
        for prompt in prompts:
            inputs = tokenizer(prompt, return_tensors="pt").to(device)

            with torch.no_grad():
                outputs = model(**inputs, output_hidden_states=True)
                # Get activation at post-instruction position
                # (typically at token index 0 of generation, or after special tokens)
                h_post = outputs.hidden_states[-1][0, -1, :]
                acts_list.append(h_post)

    refusal_acts = torch.stack(refusal_acts)
    accepted_acts = torch.stack(accepted_acts)

    # Mean difference or PCA as above
    refusal_direction = refusal_acts.mean(0) - accepted_acts.mean(0)
    refusal_direction = refusal_direction / refusal_direction.norm()

    return refusal_direction
```

## 3. Enhanced Gradient Discovery with Orthogonality

```python
from src.discovery import GradientGeometryDiscovery, GradientDiscoveryConfig

class OrthogonalityAwareDiscovery:
    """
    Extend gradient discovery to enforce orthogonality to harmfulness.
    """

    def __init__(self,
                 base_discovery: GradientGeometryDiscovery,
                 v_harmfulness: torch.Tensor,
                 lambda_ortho: float = 0.1):
        self.discovery = base_discovery
        self.v_harmfulness = v_harmfulness / v_harmfulness.norm()
        self.lambda_ortho = lambda_ortho

    def orthogonality_penalty(self, v: torch.Tensor) -> float:
        """
        Penalty for vectors aligned with harmfulness direction.
        Encourages discovered refusal vectors to be orthogonal.
        """
        v_norm = v / v.norm()
        alignment = abs((v_norm @ self.v_harmfulness).item())
        return self.lambda_ortho * alignment

    def discover_with_orthogonality_constraint(self,
                                               measurement_fn,
                                               v_init: torch.Tensor,
                                               **kwargs):
        """
        Run discovery with orthogonality constraint.
        """

        def constrained_measure_fn(v):
            # Original measurement
            R, grad = measurement_fn(v)

            # Add orthogonality penalty
            ortho_penalty = self.orthogonality_penalty(v)
            R_constrained = R - ortho_penalty

            # Modify gradient to encourage orthogonality
            v_norm = v / v.norm()
            grad_ortho = self.v_harmfulness * (v_norm @ self.v_harmfulness)
            grad_constrained = grad - self.lambda_ortho * grad_ortho

            return R_constrained, grad_constrained

        results = self.discovery.discover(
            measure_refusal_with_grad=constrained_measure_fn,
            v_init=v_init,
            **kwargs
        )

        return results
```

```python
def reply_inversion_experiment(model, tokenizer,
                              v_harmfulness, v_refusal,
                              prompts: List[str],
                              device="cuda") -> Dict:
    """
    Implement the reply inversion experiment from 2507.11878.

    Append harm-assessment question and measure:
    - Does steering harmfulness change the answer?
    - Does steering refusal change the answer?
    """

    results = {
        "harmfulness_effect": [],
        "refusal_effect": [],
        "predictions_match": []
    }

    for prompt in prompts:
        # Append harm assessment question
        prompt_with_question = (
            f"{prompt}\n"
            "Is the user's request harmful? "

        )

        # Baseline: no steering
        inputs = tokenizer(prompt_with_question, return_tensors="pt").to(device)
        with torch.no_grad():
            response_baseline = model.generate(
                inputs["input_ids"],
                max_length=20,
                temperature=0.0  # Deterministic
            )
            answer_baseline = tokenizer.decode(response_baseline[0])
            print(f"Baseline answer: {answer_baseline}")

        # Steering 1: Harmfulness direction (should flip the answer)
        # Would need to implement steering at generation time
        with torch.no_grad():
            response_harm = model.generate(
                inputs["input_ids"],
                max_length=20,
                temperature=0.0
```

## 5. Latent Guard Implementation

```python
class LatentGuard:
    """
    Safety classifier based on harmfulness direction.
    Inspired by 2507.11878's Latent Guard approach.
    """

    def __init__(self, model, tokenizer, v_harmfulness, device="cuda"):
        self.model = model
        self.tokenizer = tokenizer
        self.v_harmfulness = v_harmfulness / v_harmfulness.norm()
        self.device = device

        # Train tiny linear classifier on harmfulness direction
        self.classifier = torch.nn.Linear(1, 1)  # Single input, single output
        self.classifier.to(device)

    def train_classifier(self,
                        harmful_prompts: List[str],
                        harmless_prompts: List[str],
                        num_epochs: int = 10):
        """
        Train classifier to predict safety from harmfulness projection.
        """

        # Collect projections
        projections = []
        labels = []

        for prompts, label in [
            (harmful_prompts, 1),
            (harmless_prompts, 0)
        ]:
            for prompt in prompts:
                inputs = self.tokenizer(prompt, return_tensors="pt").to(self.device)
                with torch.no_grad():
                    outputs = self.model(**inputs, output_hidden_states=True)
                    h_inst = outputs.hidden_states[-1][0, -1, :]

                    # Project onto harmfulness direction
                    proj = (h_inst @ self.v_harmfulness).unsqueeze(0)
                    projections.append(proj)
                    labels.append(label)

        X = torch.cat(projections).detach()  # [n, 1]
        y = torch.tensor(labels, dtype=torch.float32, device=self.device)

        # Train
        optimizer = torch.optim.Adam(self.classifier.parameters(), lr=1e-3)
        loss_fn = torch.nn.BCEWithLogitsLoss()

        for epoch in range(num_epochs):
            logits = self.classifier(X)
            loss = loss_fn(logits, y.unsqueeze(1))

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            if epoch % 2 == 0:
                print(f"Epoch {epoch}: Loss = {loss.item():.4f}")

    def predict(self, prompt: str) -> Tuple[float, str]:
        """
        Predict safety of prompt.

        Returns:
            (confidence, label) where confidence in [0, 1]
            and label in ["safe", "unsafe"]
        """

        inputs = self.tokenizer(prompt, return_tensors="pt").to(self.device)

        with torch.no_grad():
            outputs = self.model(**inputs, output_hidden_states=True)
            h_inst = outputs.hidden_states[-1][0, -1, :]

            # Project onto harmfulness direction
            proj = (h_inst @ self.v_harmfulness).unsqueeze(0)

            # Classify
            logit = self.classifier(proj)
            confidence = torch.sigmoid(logit).item()

            label = "unsafe" if confidence > 0.5 else "safe"

        return confidence, label
```

```python
def measure_refusal_and_harm(v: torch.Tensor,
                            model,
                            tokenizer,
                            harm_classifier,
                            harmful_prompts,
                            harmless_prompts) -> Tuple[float, float, torch.Tensor]:
    """
    Measure both refusal rate and harmfulness perception change.

    Returns:
        (refusal_rate, harm_perception_change, gradient)
    """

    v.requires_grad = True

    # Refusal rate at generation (t_post-inst)
    refusal_rate = 0.0
    # ... (existing refusal measurement code)

    # Harmfulness perception at instruction (t_inst)
    harm_acts = []
    for prompt in harmful_prompts:
        inputs = tokenizer(prompt, return_tensors="pt")
        with torch.no_grad():
            outputs = model(**inputs, output_hidden_states=True)
            h_inst = outputs.hidden_states[-1][0, -1, :]
            harm_acts.append(h_inst)

    h_mean = torch.stack(harm_acts).mean(0)
    harm_logits = harm_classifier(h_mean)
    harm_perception = torch.sigmoid(harm_logits).mean()

    # Combined loss
    loss = refusal_rate + harm_perception
    loss.backward()

    return refusal_rate, harm_perception.item(), v.grad.clone()
```
